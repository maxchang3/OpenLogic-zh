\olsection{世界处为真}

每个模态模型都决定哪些模态!!{formula}在其中哪些世界处为真。关系「模型 $\mModel{M}$ 使!!{formula}~$!A$ 在世界~$w$ 处为真」是关系语义的基本概念。该关系按归纳方式定义；对于非模态算子，其定义与通常的真值表刻画一致。

\begin{defn}\ollabel{defn:mmodels}
  在模型~$\mModel M$ 中，\emph{!!a{formula}~$!A$ 在~$w$ 处为真}，记作
  $\mSat{M}{!A}[w]$，归纳定义如下：
  \begin{enumerate}
  \tagitem{prvFalse}{\indcase{!A}{\lfalse}{$\mSat{M}{\lfalse}[w]$ 绝不成立}。}{}
  \tagitem{prvTrue}{\indcase{!A}{\ltrue}{$\mSat{M}{\ltrue}[w]$ 恒成立}。}{}
  \item $\mSat{M}{p}[w]$ 当且仅当 $w \in V(p)$。
  \tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$ 当且仅当
    $\mSat/{M}{!B}[w]$}。}{}
  \tagitem{prvAnd}{\indcase{!A}{(!B \land !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当
    $\mSat{M}{!B}[w]$ 且 $\mSat{M}{!C}[w]$}。}{}
  \tagitem{prvOr}{\indcase{!A}{(!B \lor !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当
    $\mSat{M}{!B}[w]$ 或 $\mSat{M}{!C}[w]$}（或二者均成立）。}{}
  \tagitem{prvIf}{\indcase{!A}{(!B \lif !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当
    $\mSat/{M}{!B}[w]$ 或 $\mSat{M}{!C}[w]$}。}{}
  \tagitem{prvIff}{\indcase{!A}{(!B \liff !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当
    $\mSat{M}{!B}[w]$ 与 $\mSat{M}{!C}[w]$ 均成立，或
    $\mSat{M}{!B}[w]$ 与 $\mSat{M}{!C}[w]$ 均不成立}。}{}
  \item\ollabel{defn:sub:mmodels-box}
    \indcase{!A}{\Box !B}{$\mSat{M}{\indfrm}[w]$ 当且仅当
    对所有满足 $Rww'$ 的 $w' \in W$，都有 $\mSat{M}{!B}[w']$。}
  \item\ollabel{defn:sub:mmodels-diamond}
    \indcase{!A}{\Diamond !B}{$\mSat{M}{\indfrm}[w]$ 当且仅当
    至少存在一个满足 $Rww'$ 的 $w' \in W$，使得 $\mSat{M}{!B}[w']$。}
  \end{enumerate}
\end{defn}

注意，由条款~\olref{defn:sub:mmodels-box}可知，只要不存在满足 $Rww'$ 的~$w'$，!!a{formula}~$\Box !B$ 就在~$w$ 处为真。在这种情况下，$\Box !B$ 在~$w$ 处\emph{空真地}为真。此外，即使 $!B$ 在~$w$ 处不为真，$\Box !B$ 也可能在~$w$ 处被满足。$!B$ 在~$w$ 处为真并不能保证 $\Diamond !B$ 在~$w$ 处为真。不过，如果 $Rww$（例如 $R$ 是自反的），则可以保证这一点。如果不存在满足 $Rww'$ 的 $w'$，则对任意~$!A$，$\mSat/{M}{\Diamond !A}[w]$。

\begin{prob}
  考虑 \olref[nml][syn][rel]{fig:simple} 的模型。下列哪些成立？
    \begin{enumerate}
    \item $\mSat{M}{q}[w_1]$;
    \item $\mSat{M}{\lnot q}[w_3]$;
    \item $\mSat{M}{p \lor q}[w_1]$;
    \item $\mSat{M}{\Box (p \lor q)}[w_1]$;
    \item $\mSat{M}{\Box q}[w_3]$;
    \item $\mSat{M}{\Box \bot}[w_3]$;
    \item $\mSat{M}{\Diamond q}[w_1]$;
    \item $\mSat{M}{\Box q}[w_1]$;
    \item $\mSat{M}{\lnot \Box\Box \lnot q}[w_1]$.
    \end{enumerate}
\end{prob}

% Prove duality of \Box and \Diamond
\begin{tagblock}{prvDiamond}
\begin{prop}\ollabel{prop:dual}
  \begin{enumerate}
  \item $\mSat{M}{\Box !A}[w]$ 当且仅当 $\mSat{M}{\lnot\Diamond\lnot !A}[w]$。
  \item $\mSat{M}{\Diamond !A}[w]$ 当且仅当 $\mSat{M}{\lnot\Box\lnot !A}[w]$。
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item 由 $\mSat{M}{}[w]$ 的定义，$\mSat{M}{\lnot\Diamond\lnot !A}[w]$ 当且仅当
      $\mSat/{M}{\Diamond\lnot !A}[w]$。$\mSat{M}{\Diamond\lnot
      !A}[w]$ 当且仅当存在某个满足 $Rww'$ 的 $w'$，$\mSat{M}{\lnot
      !A}[w']$。因此，$\mSat/{M}{\Diamond\lnot !A}[w]$ 当且仅当对所有
    满足 $Rww'$ 的 $w'$，$\mSat/{M}{\lnot !A}[w']$。我们还有
    $\mSat/{M}{\lnot !A}[w']$ 当且仅当 $\mSat{M}{!A}[w']$。综合起来我们有
    $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ 当且仅当对所有满足 $Rww'$ 的 $w'$，
    $\mSat{M}{!A}[w']$。再次由 $\mSat{M}{}[w]$ 的定义，此即
      $\mSat{M}{\Box !A}[w]$ 成立。
    \item \iftag{probDiamond}{习题。}{$\mSat{M}{\lnot\Box\lnot !A}[w]$ 当且仅当
      $\mSat/{M}{\Box\lnot !A}$。$\mSat{M}{\Box\lnot !A}[w]$ 当且仅当对所有
      满足 $Rww'$ 的 $w'$，$\mSat{M}{\lnot !A}[w']$。因此，
      $\mSat/{M}{\Box\lnot !A}[w]$ 当且仅当存在某个满足 $Rww'$ 的 $w'$，
      $\mSat/{M}{\lnot !A}[w']$。我们还有 $\mSat/{M}{\lnot
        !A}[w']$ 当且仅当 $\mSat{M}{!A}[w']$。综合起来我们有
      $\mSat{M}{\lnot\Box\lnot !A}[w]$ 当且仅当存在某个满足
      $Rww'$ 的~$w'$，$\mSat{M}{!A}[w']$。再次由 $\mSat{M}{}[w]$
      的定义，此即 $\mSat{M}{\Diamond !A}[w]$ 成立。}
  \end{enumerate}
\end{proof}

\begin{probtag}{probDiamond}
  补全 \olref[nml][syn][trw]{prop:dual} 的证明。
\end{probtag}
\end{tagblock}

\begin{prob}
  设 $\mModel{M} = \tuple{W, R, V}$ 是一个模型，并设 $w_1, w_2 \in
  W$ 满足：
  \begin{enumerate}
  \item 当且仅当 $w_2 \in V(p)$ 时 $w_1 \in V(p)$（对每个!!{propositional variable} $p$）；且
  \item 对所有 $w \in W$：$Rw_1w$ 当且仅当 $Rw_2w$。
  \end{enumerate}
  用!!{formula}上的归纳证明：对所有!!{formula} $!A$，
  $\mSat{M}{!A}[w_1]$ 当且仅当 $\mSat{M}{!A}[w_2]$。
\end{prob}

\begin{probtag}{prvDiamond}
  设 $\mModel{M} = \tuple{W, R, V}$。证明
  $\mSat{M}{\lnot\Diamond !A}[w]$ 当且仅当
  $\mSat{M}{\Box\lnot!A}[w]$。
\end{probtag}
