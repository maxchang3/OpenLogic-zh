\olsection{\Log{K} 的完全性}

\begin{explain}
  要证明!!{tableau}方法是完全的，我们必须证明：当没有闭的!!{tableau}可用
  以表明 $\Gamma \Proves !A$ 时，就有 $\Gamma \Entails/ !A$，即存在一个
  反模型。但「没有闭的!!{tableau}」意味着我们尝试构造一个的所有方式都必然
  不能闭合。关键在于看出：如果每一种这样的方式都不能闭合，那么一种特定的、
  \emph{系统的且穷尽的}方式也不能闭合。而若闭的!!{tableau}存在，那么这种
  系统且穷尽的方式就会闭合。单一的若干表列会在其开放分支之中，包含定义一个
  反模型所需的全部信息。由这个表列中的一条开放分支给出的反模型，会把该分支上
  用到的所有前缀当作世界，并且!!a{propositional variable}~$p$ 在 $\sigma$ 处为真，
  当且仅当 $\sFmla{\True}{p}[\sigma]$ 出现在该分支上。
\end{explain}

\begin{defn}
  若 !!a{tableau} 中的一条分支，只要它包含一个可以对其应用某规则的带前缀的!!{formula} $\sFmla{S}{!A}[\sigma]$，它就会包含
  \begin{enumerate}
    \item 对于命题堆叠规则的情形，该规则相应的结论带前缀!!{formula}；
    \item 对于命题分支规则的情形，相应的结论!!{formula}之一；
    \item 对于要求新前缀的模态规则的情形，至少一个可能的结论；
    \item 对于要求已用前缀的模态规则的情形，对分支上出现的每个前缀都给出相应
      的结论。
  \end{enumerate}
  则称该分支是完成的。
\end{defn}

\begin{explain}
  例如，一条完成的分支每当包含 $\sFmla{\True}{!B
  \land !C}$ 时，就会包含 $\sFmla{\True}{!B}[\sigma]$
  和 $\sFmla{\True}{!C}[\sigma]$。若它包含 $\sFmla{\True}{!B \lor !C}[\sigma]$，则
  它至少包含 $\sFmla{\False}{!B}[\sigma]$ 和
  $\sFmla{\True}{!C}[\sigma]$ 之一。若它包含 \iftag{prvBox}
  {$\sFmla{\False}{\Box}[\sigma]$} {$\sFmla{\True}{\Diamond}[\sigma]$}，
  那么对至少一个~$n$，它也包含 \iftag{prvBox} {$\sFmla{\False}{\Box}[\sigma.n]$}
  {$\sFmla{\True}{\Diamond}[\sigma.n]$}。而且，每当它包含 \iftag{prvBox}
  {$\sFmla{\True}{\Box}[\sigma]$}
  {$\sFmla{\False}{\Diamond}[\sigma]$} 时，对用于该分支的每一个~$n$（即每个满足
  $\sigma.n$ 被用于该分支的 $n$），它也包含 \iftag{prvBox}
  {$\sFmla{\True}{\Box}[\sigma.n]$}
  {$\sFmla{\False}{\Diamond}[\sigma.n]$}。
\end{explain}

\begin{prop}\ollabel{prop:complete-tableau}
  每个有穷的 $\Gamma$ 都有一个所有分支均是完成分支的!!a{tableau}。
\end{prop}

\begin{proof}
  考虑 ~$\Gamma$ 的一个!!a{tableau}中的一条开放分支。分支中有有穷多个可对其应用规则的
  带前缀!!{formula}。按某一固定顺序（例如从上到下），对于满足条件 (1)--(4) 尚
  未成立的那些带前缀!!{formula}中的每一个，都应用可应用于它的规则来延伸分支。
  在某些情形这将导致分支；对所得每条分支的末端，对所有剩余的带前缀!!{formula}应用规则。由于带前缀!!{formula}的数目是有穷的，且分支上已用前缀的
  数目也是有穷的，这一过程最终将得出（可能许多条）延伸原分支的分支。对每条分支
  应用这一过程，并重复进行。但据构造，每条分支都是闭的。
\end{proof}

\begin{thm}[完全性]
  \ollabel{thm:tableau-completeness}
  若 $\Gamma$ 没有闭的!!{tableau}，则 $\Gamma$ 是可满足的。
\end{thm}

\begin{proof}
  由该命题，$\Gamma$ 有一个所有分支均为完成分支的!!a{tableau}。由于 $\Gamma$ 没有闭的!!{tableau}，则存在一个其中至少有一条开放且完成的分支的!!a{tableau}。设 $\Delta$ 为分支上带前缀的!!{formula}的集合，$P(\Delta)$ 为出现在其中的前缀的集合。

  我们定义一个模型~$\mModel{M(\Delta)} = \tuple{P(\Delta), R, V}$，其中世界就是
  出现在~$\Delta$ 中的前缀，可及关系由下式给出：
\[
R\sigma\sigma' \quad \text{当且仅当} \quad
\sigma'=\sigma.n \quad \text{对某个~$n$}
\]
以及
\[
V(p) = \Setabs{\sigma}{\sFmla{\True}{p}[\sigma] \in \Delta}.
\]
  我们通过对~$!A$ 作归纳来证明：若 $\sFmla{\True}{!A}[\sigma] \in
\Delta$，则 $\mSat{M(\Delta)}{!A}[\sigma]$；而若
$\sFmla{\False}{!A}[\sigma] \in \Delta$，则
$\mSat/{M(\Delta)}{!A}[\sigma]$。

\begin{enumerate}
  \item \indcase{!A}{p}{若 $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$，则
    $\sigma \in V(p)$（由~$V$ 的定义）从而
    $\mSat{M(\Delta)}{\indfrm}[\sigma]$。

    若 $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$，则
    $\sFmla{\True}{\indfrm}[\sigma] \notin \Delta$，因为否则该分支就会是闭的。故
    $\sigma \notin V(p)$，从而
    $\mSat/{M(\Delta)}{\indfrm}[\sigma]$。}
  \item \indcase{!A}{\lnot !B}{\iftag{probNot}{习题。}{若
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$，则由于分支是完成的，有
      $\sFmla{\False}{!B}[\sigma] \in \Delta$。由归纳假设，有
      $\mSat/{M(\Delta)}{!B}[\sigma]$，从而
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$。

      若 $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$，则由于分支是完成的，有
      $\sFmla{\True}{!B}[\sigma] \in \Delta$。由归纳假设，有
      $\mSat{M(\Delta)}{!B}[\sigma]$，从而
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$。}}
  \item \indcase{!A}{!B \land !C}{\iftag{probAnd}{习题。}{若
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$，则由于分支是完成的，会有
      $\sFmla{\True}{!B}[\sigma] \in \Delta$ 且
      $\sFmla{\True}{!C}[\sigma] \in \Delta$。由归纳假设，有
      $\mSat{M(\Delta)}{!B}[\sigma]$ 且
      $\mSat{M(\Delta)}{!C}[\sigma]$。从而
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$。

      若 $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$，则由于分支是完成的，有
      $\sFmla{\False}{!B}[\sigma] \in \Delta$ 或
      $\sFmla{\False}{!C}[\sigma] \in \Delta$。由归纳假设，有
      $\mSat/{M(\Delta)}{!B}[\sigma]$ 或
      $\mSat/{M(\Delta)}{!B}[\sigma]$。从而
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$。}}
  \item \indcase{!A}{!B \lor !C}{\iftag{probOr}{习题。}{若
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$，则由于分支是完成的，有
      $\sFmla{\True}{!B}[\sigma] \in \Delta$ 或
      $\sFmla{\True}{!C}[\sigma] \in \Delta$。由归纳假设，有
      $\mSat{M(\Delta)}{!B}[\sigma]$ 或
      $\mSat{M(\Delta)}{!C}[\sigma]$。从而
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$。

      若 $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$，则由于分支是完成的，有
      $\sFmla{\False}{!B}[\sigma] \in \Delta$ 且
      $\sFmla{\False}{!C}[\sigma] \in \Delta$。由归纳假设，有
      $\mSat/{M(\Delta)}{!B}[\sigma]$ 且
      $\mSat/{M(\Delta)}{!B}[\sigma]$。从而
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$。}}
  \item \indcase{!A}{!B \lif !C}{\iftag{probIf}{习题。}{若
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$，则由于分支是完成的，有
      $\sFmla{\False}{!B}[\sigma] \in \Delta$ 或
      $\sFmla{\True}{!C}[\sigma] \in \Delta$。由归纳假设，有
      $\mSat/{M(\Delta)}{!B}[\sigma]$ 或
      $\mSat{M(\Delta)}{!C}[\sigma]$。从而
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$。

      若 $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$，则由于分支是完成的，有
      $\sFmla{\True}{!B}[\sigma] \in \Delta$ 且
      $\sFmla{\False}{!C}[\sigma] \in \Delta$。由归纳假设，有
      $\mSat{M(\Delta)}{!B}[\sigma]$ 且
      $\mSat/{M(\Delta)}{!B}[\sigma]$。从而
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$。}}
  \item \indcase{!A}{\Box !B}{\iftag{probBox}{习题。}{若
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$，则因分支是完成的，对用于分支的
      每一个 $\sigma.n$，即对 $P(\Delta)$ 中每个满足 $R\sigma\sigma'$ 的 $\sigma'$，
      有 $\sFmla{\True}{!B}[\sigma.n] \in \Delta$。由归纳假设，对每个满足
      $R\sigma\sigma'$ 的 $\sigma'$，有 $\mSat{M(\Delta)}{!B}[\sigma']$。因此，
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$。

      若 $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$，则因分支是完成的，对某个
      $\sigma.n$，有 $\sFmla{\False}{!B}[\sigma.n] \in \Delta$。由归纳假设，有
      $\mSat/{M(\Delta)}{!B}[\sigma.n]$。由于 $R\sigma(\sigma.n)$，
      存在某个~$\sigma'$ 使得
      $\mSat/{M(\Delta)}{!B}[\sigma']$。从而
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$。}}
  \item \indcase{!A}{\Diamond !B}{\iftag{probDiamond}{习题。}{若
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$，则因分支是完成的，对某个
      $\sigma.n$，有 $\sFmla{\True}{!B}[\sigma.n] \in \Delta$。由归纳假设，有
      $\mSat{M(\Delta)}{!B}[\sigma.n]$。由于 $R\sigma(\sigma.n)$，
      存在某个~$\sigma'$ 使得
      $\mSat{M(\Delta)}{!B}[\sigma']$。从而
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$。

      若 $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$，则因分支是完成的，对用于分支的
      每一个 $\sigma.n$，即对 $P(\Delta)$ 中每个满足 $R\sigma\sigma'$ 的 $\sigma'$，
      有 $\sFmla{\False}{!B}[\sigma.n] \in \Delta$。由归纳假设，对每个满足
      $R\sigma\sigma'$ 的 $\sigma'$，有 $\mSat/{M(\Delta)}{!B}[\sigma']$。因此，
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$。}}
\end{enumerate}
由于 $\Gamma \subseteq \Delta$，有 $\mSat{M(\Delta)}{\Gamma}$。
\end{proof}

\begin{prob}
  完成 \olref[nml][tab][cpl]{thm:tableau-completeness} 的证明。
\end{prob}

\begin{cor}
\ollabel{cor:entailment-completeness}
若 $\Gamma \Entails !A$，则 $\Gamma \Proves !A$。
\end{cor}

\begin{cor}
\ollabel{cor:weak-completeness}
若 $!A$ 在一切模型中为真，则 $\Proves !A$。
\end{cor}
