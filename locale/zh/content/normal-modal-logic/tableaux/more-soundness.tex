\olsection{附加规则的可靠性}

我们说，一个规则对某一类模型是可靠的，如果每当!!a{tableau}中的一个支在这类模型中的某个模型里是可满足的，那么施用该规则所得的支也在这类模型中的某个模型里是可满足的。

\begin{prop}
  \ollabel{prop:soundness-T}
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ 和 }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{}{}%
  对自反的模型是可靠的。
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{将 T$\Box$ 施用于 $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$ 来扩张这个支：
  \iftag{probBox}{习题。}{这会在支上产生一个新的!!{signed
      formula}~$\sFmla{\True}{!B}[\sigma]$。设
    $\mSat{M}{\Gamma}[f]$，特别地，$\mSat{M}{\Box
      !B}[f(\sigma)]$。由于 $R$ 是自反的，我们知道
    $Rf(\sigma)f(\sigma)$。因此，$\mSat{M}{!B}[f(\sigma)]$，即
    $\mModel{M}, f$ 满足 $\sFmla{\True}{!B}[\sigma]$。}}{}
\tagitem{prvDiamond}{将 T$\Diamond$ 施用于 $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$ 来扩张这个支：
  \iftag{probDiamond}{习题。}{这会在支上产生一个新的!!{signed
      formula}~$\sFmla{\False}{!B}[\sigma]$。设
    $\mSat{M}{\Gamma}[f]$，特别地，$\mSat/{M}{\Diamond
      !B}[f(\sigma)]$。由于 $R$ 是自反的，我们知道
    $Rf(\sigma)f(\sigma)$。因此，$\mSat/{M}{!B}[f(\sigma)]$，即
    $\mModel{M}, f$ 满足 $\sFmla{\False}{!B}[\sigma]$。}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  完成 \olref[nml][tab][msn]{prop:soundness-T} 的证明。
\end{prob}
\tagendprob

  \iftag{prvDiamond}{\Ax{4}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{}{}%
  对传递的模型是可靠的。
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
