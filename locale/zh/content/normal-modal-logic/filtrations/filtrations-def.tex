\olsection{过滤}

与其定义 $\mModel{M}$ 经过 $\Gamma$ 的「那个」过滤，我们不如定义何时一个模型~$\mModel{M^*}$ 算作 $\mModel{M}$ 的一个过滤。所有过滤都有相同的世界集~$W^*$ 和相同的赋值~$V^*$。但不同的过滤可能有不同的可及关系~$R^*$。然而，要算作一个过滤，$R^*$ 必须满足若干条件。这些条件正是我们要用来证明主要结果所需的东西，即，只要 $!A \in \Gamma$，就有
$\mSat{M}{!A}[w]$ 当且仅当 $\mSat{M^*}{!A}[{[w]}]$。

\begin{defn}\ollabel{defn:filtration}
  设 $\Gamma$ 对子公式封闭，且 $\mModel{M} = \tuple{W, R, V}$。$\mModel{M}$ 经过 $\Gamma$ 的\emph{过滤}是任何模型 $\mModel{M^*} = \tuple{W^*,R^*,V^*}$，其中：
  \begin{enumerate}
  \item $W^* = \Setabs{[w]}{w \in W}$；
  \item \ollabel{defn:filtration-R}%
    对任意 $u,v \in W$：
    \begin{enumerate}
    \item \ollabel{defn:filtration-R1}%
      若 $Ruv$，则 $R^*[u][v]$；
    \tagitem{prvBox}{\ollabel{defn:filtration-R2}%
      若 $R^*[u][v]$，则对任意 $\Box !A \in \Gamma$，若
      $\mSat{M}{\Box!A}[u]$，则 $\mSat{M}{!A}[v]$}{};
    \tagitem{prvDiamond}{\ollabel{defn:filtration-R3}%
      若 $R^*[u][v]$，则对任意 $\Diamond !A \in \Gamma$，若
      $\mSat{M}{!A}[v]$，则 $\mSat{M}{\Diamond!A}[u]$。}{}
    \end{enumerate}
  \item $V^*(p) = \Setabs{[u]}{u \in V(p)}$。
  \end{enumerate}
\end{defn}

值得想一想 $V^*(p)$ 是什么：它是由~$\mModel{M}$ 中 $p$ 为真之所有世界~$w$ 的等价类~$[w]$ 组成的集合。一方面，若 $w \in V(p)$，则该定义保证 $[w] \in V^*(p)$。然而，若 $[w] \in V^*(p)$，则并不必然有 $w \in V(p)$。若 $[w] \in V^*(p)$，我们只能保证 $[w] = [u]$ 对于\emph{某个}~$u \in V(p)$ 成立。当然，$[w] = [u]$ 意味着 $w \equiv u$。因此，当 $[w] \in V^*(p)$ 时，我们（只能）推出，对某个 $u \in V(p)$ 有 $w \equiv u$。

\begin{thm}\ollabel{thm:filtrations}
  若 $\mModel{M^*}$ 是 $\mModel{M}$ 经过 $\Gamma$ 的一个过滤，则对每个 $!A \in \Gamma$ 和 $w \in W$，有 $\mSat{M}{!A}[w]$ 当且仅当 $\mSat{M^*}{!A}[{[w]}]$。
\end{thm}

\begin{proof}
  对 $!A$ 作归纳，利用 $\Gamma$ 对子公式封闭这一事实。由于 $!A \in \Gamma$ 且 $\Gamma$ 对子!!{formula}封闭，$!A$ 的所有子!!{formula}也都 $\in \Gamma$。因此在每一步归纳步骤中，归纳假设都适用于 $!A$ 的子!!{formula}。
  \begin{enumerate}
    \tagitem{prvFalse}{\indcase{!A}{\lfalse}{既没有
        $\mSat{M}{\indfrm}[w]$，也没有 $\mSat{M^*}{\indfrm}[{[w]}]$。}}{}
    \tagitem{prvTrue}{\indcase{!A}{\ltrue}{既有
        $\mSat{M}{\indfrm}[w]$，也有 $\mSat{M^*}{\indfrm}[{[w]}]$。}}{}
    \item \indcase{!A}{p}{从左到右的方向是直接的，因为
      $\mSat{M}{\indfrm}[w]$ 仅当 $w \in V(p)$，这意味着 $[w]
      \in V^*(p)$，即 $\mSat{M^*}{\indfrm}[{[w]}]$。反之，
      设 $\mSat{M^*}{\indfrm}[{[w]}]$，即 $[w] \in
      V^*(p)$。于是对某个 $v \in V(p)$，$w \equiv v$。当然
      此时也有 $\mSat{M}{p}[v]$。由于 $w \equiv v$，$w$ 和 $v$ 使来自~$\Gamma$ 的相同!!{formula}为真。由于根据假设 $p
      \in \Gamma$ 且 $\mSat{M}{p}[v]$，故 $\mSat{M}{\indfrm}[w]$。}

    \tagitem{prvNot}{\iftag{probNot}{习题。}{\indcase{!A}{\lnot
          !B}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat/{M}{!B}[w]$。根据
          归纳假设，$\mSat/{M}{!B}[w]$ 当且仅当
          $\mSat/{M^*}{!B}[{[w]}]$。最后，$\mSat/{M^*}{!B}[{[w]}]$
          当且仅当 $\mSat{M^*}{\indfrm}[{[w]}]$。}}}{}

    \tagitem{prvAnd}{\iftag{probAnd}{习题。}{\indcase{!A}{(!B \land
          !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat{M}{!B}[w]$ 且
          $\mSat{M}{!C}[w]$。根据归纳假设，$\mSat{M}{!B}[w]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$，且 $\mSat{M}{!C}[w]$ 当且仅当
          $\mSat{M^*}{!C}[{[w]}]$。而 $\mSat{M^*}{\indfrm}[{[w]}]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$ 且
          $\mSat{M^*}{!C}[{[w]}]$。}}}{}

    \tagitem{prvOr}{\iftag{probOr}{习题。}{\indcase{!A}{(!B \lor
          !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat{M}{!B}[w]$ 或
          $\mSat{M}{!C}[w]$。根据归纳假设，$\mSat{M}{!B}[w]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$，且 $\mSat{M}{!C}[w]$ 当且仅当
          $\mSat{M^*}{!C}[{[w]}]$。而 $\mSat{M^*}{\indfrm}[{[w]}]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$ 或
          $\mSat{M^*}{!C}[{[w]}]$。}}}{}

    \tagitem{prvIf}{\iftag{probIf}{习题。}{\indcase{!A}{(!B \lif
          !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat/{M}{!B}[w]$ 或
          $\mSat{M}{!C}[w]$。根据归纳假设，$\mSat{M}{!B}[w]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$，且 $\mSat{M}{!C}[w]$ 当且仅当
          $\mSat{M^*}{!C}[{[w]}]$。而 $\mSat{M^*}{\indfrm}[{[w]}]$
          当且仅当 $\mSat/{M^*}{!B}[{[w]}]$ 或
          $\mSat{M^*}{!C}[{[w]}]$。}}}{}

    \tagitem{prvIff}{\iftag{probIff}{习题。}{\indcase{!A}{(!B \liff
          !C)}{$\mSat{M}{\indfrm}[w]$ 当且仅当 $\mSat{M}{!B}[w]$ 且
          $\mSat{M}{!C}[w]$，或 $\mSat/{M}{!B}[w]$ 且
          $\mSat/{M}{!C}[w]$。根据归纳假设，$\mSat{M}{!B}[w]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$，且 $\mSat{M}{!C}[w]$ 当且仅当
          $\mSat{M^*}{!C}[{[w]}]$。而 $\mSat{M^*}{\indfrm}[{[w]}]$
          当且仅当 $\mSat{M^*}{!B}[{[w]}]$ 且
          $\mSat{M^*}{!C}[{[w]}]$，或 $\mSat/{M^*}{!B}[{[w]}]$ 且
          $\mSat/{M^*}{!C}[{[w]}]$。}}}{}

    \tagitem{prvBox}{\iftag{probBox}{习题。}{\indcase{!A}{\Box
          !B}{设 $\mSat{M}{\indfrm}[w]$；要证明
          $\mSat{M^*}{\indfrm}[{[w]}]$，令 $v$ 使得
          $R^*[w][v]$。由
          \olref{defn:filtration}\olref{defn:filtration-R2}，我们有
          $\mSat{M}{!B}[v]$，且根据归纳假设
          $\mSat{M^*}{!B}[{[v]}]$。由于 $v$ 是任意的，
          $\mSat{M^*}{\indfrm}[{[w]}]$ 随之成立。

          反之，设 $\mSat{M^*}{\indfrm}[{[w]}]$，并令 $v$
          任意，使得 $Rwv$。由
          \olref{defn:filtration}\olref{defn:filtration-R1}，我们有
          $R^*[w][v]$，从而 $\mSat{M^*}{!B}[{[v]}]$；根据归纳
          假设 $\mSat{M}{!B}[v]$，且由于 $v$ 是任意的，
          $\mSat{M}{\indfrm}[w]$。}}}{}

    \tagitem{prvDiamond}{\iftag{probDiamond}{习题。}{\indcase{!A}{\Diamond
          !B}{设 $\mSat{M}{\indfrm}[w]$。则对某个 $v \in W$，
          $Rwv$ 且 $\mSat{M}{!B}[v]$。根据归纳假设
          $\mSat{M^*}{!B}[{[v]}]$，且由
          \olref{defn:filtration}\olref{defn:filtration-R1}，我们有
          $R^*[w][v]$。因此，$\mSat{M^*}{\indfrm}[{[w]}]$。

          现在设 $\mSat{M^*}{\indfrm}[{[w]}]$。则对某个满足 $R^*[w][v]$ 的 $[v]
          \in W^*$，$\mSat{M^*}{!B}[{[v]}]$。根据
          归纳假设 $\mSat{M}{!B}[v]$。由
          \olref{defn:filtration}\olref{defn:filtration-R3}，我们有
          $\mSat{M}{\indfrm}[w]$。}}}{}
  \end{enumerate}
\end{proof}

\begin{prob}
  完成 \olref[nml][fil][fil]{thm:filtrations} 的证明
\end{prob}

在一个模型中对世界处的真成立的性质，也适用于在模型中的真和在模型类上的有效性。

\begin{cor}
  设 $\Gamma$ 对子公式封闭。则：
  \begin{enumerate}
  \item 若 $\mModel{M^*}$ 是 $\mModel{M}$ 经过 $\Gamma$ 的一个过滤，则对任意 $!A \in \Gamma$：$\mSat{M}{!A}$ 当且仅当 $\mSat{M^*}{!A}$。
  \item 若 $\mClass{C}$ 是一个模型类，且 $\Gamma(\mClass{C})$
    是 $\mClass{C}$ 中模型的 $\Gamma$-过滤组成的类，则任何!!{formula}~$!A \in \Gamma$ 在 $\mClass{C}$ 上有效当且仅当它在 $\Gamma(\mClass{C})$ 上有效。
  \end{enumerate}
\end{cor}
