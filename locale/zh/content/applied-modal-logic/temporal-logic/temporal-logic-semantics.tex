\olsection{时态逻辑的语义}

\begin{defn}
时态逻辑的基础语言包含
\begin{enumerate}
  \tagitem{prvFalse}{用于表示!!{falsity}~$\lfalse$ 的命题常元。}{}
  \tagitem{prvTrue}{用于表示!!{truth}~$\ltrue$ 的命题常元。}{}
  \item 一个由!!{propositional variable}组成的!!{denumerable}集合：$\Obj
    p_0$、$\Obj p_1$、$\Obj p_2$、\dots
  \item 命题联结词：\startycommalist
  \iftag{prvNot}{\ycomma $\lnot$（否定）}{}%
  \iftag{prvAnd}{\ycomma $\land$（合取）}{}%
  \iftag{prvOr}{\ycomma $\lor$（析取）}{}%
  \iftag{prvIf}{\ycomma $\lif$（!!{conditional}）}{}%
  \iftag{prvIff}{\ycomma $\liff$（!!{biconditional}）}{}.
  \item 过去算子 $\Ptemp$ 和 $\Htemp$。
  \item 未来算子 $\Ftemp$ 和 $\Gtemp$。
\end{enumerate}
\end{defn}

稍后，我们将讨论可能添加其他类型的模态算子。

\begin{defn}
时态语言的\emph{!!^{formula}s} 归纳定义如下：
\begin{enumerate}
\tagitem{prvFalse}{$\lfalse$ 是一个原子!!{formula}。}{}

\tagitem{prvTrue}{$\ltrue$ 是一个原子!!{formula}。}{}

\item 每个命题变元 $\Obj p_i$ 都是一个（原子的）!!{formula}。

\tagitem{prvNot}{如果 $!A$ 是!!a{formula}，则 $\lnot !A$ 是!!a{formula}。}{}

\tagitem{prvAnd}{如果 $!A$ 和 $!B$ 是!!{formula}，则 $(!A \land
  !B)$ 是!!a{formula}。}{}

\tagitem{prvOr}{如果 $!A$ 和 $!B$ 是!!{formula}，则 $(!A \lor !B)$
  是!!a{formula}。}{}

\tagitem{prvIf}{如果 $!A$ 和 $!B$ 是!!{formula}，则 $(!A \lif !B)$
  是!!a{formula}。}{}

\tagitem{prvIff}{如果 $!A$ 和 $!B$ 是!!{formula}，则 $(!A \liff !B)$
  是!!a{formula}。}{}

\item 如果 $!A$ 是!!a{formula}，则 $\Ptemp  !A$、$\Htemp  !A$、$F !A$、$\Gtemp  !A$ 都是!!{formula}。

\tagitem{limitClause}{其他任何东西都不是!!a{formula}。}{}
\end{enumerate}
\end{defn}

时态逻辑的语义与其他类型的内涵逻辑一样，都以关系模型来给出。

\begin{defn}
  时态语言的\emph{模型}是一个三元组
  $\mModel{M} = \tuple{T, \prec, V}$，其中
  \begin{enumerate}
  \item $T$ 是一个非空集合，可解释为时间中的点。
  \item $\prec$ 是 $T$ 上的一个二元关系。
  \item $V$ 是一个函数，它把每个!!{propositional
    variable}~$p$ 指派为一个由时间中的点组成的集合 $V(p)$。
  \end{enumerate}
  当 $t \prec t'$ 成立时，我们说 $t$ \emph{先于}~$t'$。
  当 $t \in V(p)$ 时，我们说 $p$ 在~$t$ 处\emph{为真}。
\end{defn}

目前，你会注意到我们对先于关系 $\prec$ 没有强加任何条件。这意味着目前对我们的时态模型的结构没有任何限制，因此我们可能有时间呈线性、分叉、环形，或者具有任何结构的模型。

正如正规模态逻辑一样，每个时态模型都决定哪些!!{formula}在其中的哪些点处算作真。我们对关系语义的基本概念使用相同的记号
「模型 $\mModel{M}$ 在点~$t$ 处使!!{formula}~$!A$ 为真」。该关系
是归纳定义的，且对一切非模态算子而言与正规模态情形相同。

\begin{defn}\ollabel{defn:tmodels}
  \emph{!!a{formula}~$!A$ 在~$t$ 处的真}于~$\mModel M$，记为
  $\mSat{M}{!A}[t]$，归纳定义如下：
  \begin{enumerate}
  \tagitem{prvFalse}{\indcase{!A}{\lfalse}{绝不有 $\mSat{M}{\lfalse}[t]$。}}{}
  \tagitem{prvTrue}{\indcase{!A}{\ltrue}{总有 $\mSat{M}{\ltrue}[t]$。}}{}
  \item $\mSat{M}{p}[t]$ 当且仅当 $t \in V(p)$
  \tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[t]$ 当且仅当
    $\mSat/{M}{!B}[t]$。}}{}
  \tagitem{prvAnd}{\indcase{!A}{(!B \land !C)}{$\mSat{M}{\indfrm}[t]$ 当且仅当
    $\mSat{M}{!B}[t]$ 且 $\mSat{M}{!C}[t]$。}}{}
  \tagitem{prvOr}{\indcase{!A}{(!B \lor !C)}{$\mSat{M}{\indfrm}[t]$ 当且仅当
    $\mSat{M}{!B}[t]$ 或 $\mSat{M}{!C}[t]$}（或两者）。}{}
  \tagitem{prvIf}{\indcase{!A}{(!B \lif !C)}{$\mSat{M}{\indfrm}[t]$ 当且仅当
    $\mSat/{M}{!B}[t]$ 或 $\mSat{M}{!C}[t]$。}}{}
  \tagitem{prvIff}{\indcase{!A}{(!B \liff !C)}{$\mSat{M}{\indfrm}[t]$ 当且仅当
    要么 $\mSat{M}{!B}[t]$ 与 $\mSat{M}{!C}[t]$ 两者都成立，要么
    $\mSat{M}{!B}[t]$ 与 $\mSat{M}{!C}[t]$ 两者都不成立。}}{}
  \item\ollabel{defn:sub:mmodels-p}
    \indcase{!A}{\Ptemp  !B}{$\mSat{M}{\indfrm}[t]$ 当且仅当
    $\mSat{M}{!B}[t']$ 对某个满足 $t' \prec t$ 的 $t' \in T$ 成立}
\item\ollabel{defn:sub:mmodels-h}
    \indcase{!A}{\Htemp  !B}{$\mSat{M}{\indfrm}[t]$ 当且仅当
    $\mSat{M}{!B}[t']$ 对每个满足 $t' \prec t$ 的 $t' \in T$ 都成立}
   \item\ollabel{defn:sub:mmodels-f}
    \indcase{!A}{\Ftemp  !B}{$\mSat{M}{\indfrm}[t]$ 当且仅当
    $\mSat{M}{!B}[t']$ 对某个满足 $t \prec t'$ 的 $t' \in T$ 成立}
\item\ollabel{defn:sub:mmodels-g}
    \indcase{!A}{\Gtemp  !B}{$\mSat{M}{\indfrm}[t]$ 当且仅当
    $\mSat{M}{!B}[t']$ 对每个满足 $t \prec t'$ 的 $t' \in T$ 都成立}
  \end{enumerate}
\end{defn}

基于该语义，你可能能够看出算子 $\Ptemp$ 与~$\Htemp$ 是互相对偶的，算子 $\Ftemp$ 与~$\Gtemp$ 也是这样，因此我们能把 $\Htemp  !A$ 定义为 $\lnot \Ptemp \lnot !A$，$\Gtemp$ 与~$\Ftemp$ 同理。
