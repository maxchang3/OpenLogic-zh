\olsection{\usetoken{P}{valuation}与满足}

\begin{defn}[!!^{valuation}s]
设 $\{\True, \False\}$ 是「真」（「true」）与「假」（「false」）这两个真值构成的集合。$\Lang{L_0}$ 的一个 \emph{!!{valuation}} 是把 $\True$ 或 $\False$ 指派给语言中每个!!{propositional variable}的函数~$\pAssign{v}$，即 $\pAssign{v} \colon
\PVar \to \{\True, \False \}$。
\end{defn}

\begin{defn}
  给定!!a{valuation}~$\pAssign{v}$，归纳地定义求值函数
  $\pValue{v} \colon \Frm[L_0] \to \{\True, \False \}$：
  \begin{align*}
    \iftag{prvFalse}{\pValue{v}(\lfalse) & = \False; \\}{}
    \iftag{prvTrue}{\pValue{v}(\ltrue) & = \True; \\}{}
    \pValue{v}(\Obj p_n) & = \pAssign{v}(\Obj p_n); \\
    \iftag{prvNot}{\pValue{v}(\lnot !A) & = \begin{cases}
        \True & \text{如果 } \pValue{v}(!A) = \False;\\
        \False & \text{否则。}
      \end{cases} \\ }{}
    \iftag{prvAnd}{\pValue{v}(!A \land !B) & = \begin{cases}
        \True &
        \text{如果 $\pValue{v}(!A) = \True$ 且 $\pValue{v}(!B) = \True$;}\\
        \False &
        \text{如果 $\pValue{v}(!A) = \False$ 或 $\pValue{v}(!B) = \False$}.
      \end{cases}\\}{}
    \iftag{prvOr}{\pValue{v}(!A \lor !B) & = \begin{cases}
        \True &
        \text{如果 $\pValue{v}(!A) = \True$ 或 $\pValue{v}(!B) = \True$;}\\
        \False &
        \text{如果 $\pValue{v}(!A) = \False$ 且 $\pValue{v}(!B) = \False$}.
      \end{cases}\\}{}
    \iftag{prvIf}{\pValue{v}(!A \lif !B) & = \begin{cases}
        \True &
        \text{如果 $\pValue{v}(!A) = \False$ 或 $\pValue{v}(!B) = \True$;}\\
        \False &
        \text{如果 $\pValue{v}(!A) = \True$ 且 $\pValue{v}(!B) = \False$}.
      \end{cases}\\}{}
    \iftag{prvIff}{\pValue{v}(!A \liff !B) & = \begin{cases}
        \True &
        \text{如果 $\pValue{v}(!A) = \pValue{v}(!B)$;}\\
        \False &
        \text{如果 $\pValue{v}(!A) \neq \pValue{v}(!B)$}.
      \end{cases}}{}
\end{align*}
\end{defn}

\begin{explain}
这些条目对应于如下的真值表：
\begin{center}
\iftag{prvNot}{
  \begin{tabular}{|c||c|} \hline
    $!A$ & $ \lnot !A$ \\
    \hline \hline
    $\True$ & $\False$ \\
    $\False$ & $\True$ \\
    \hline
  \end{tabular}
}{}
\iftag{prvAnd}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \land !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\False$ \\
    $\False$ & $\True$ & $\False$ \\
    $\False$ & $\False$ & $\False$ \\
    \hline
  \end{tabular}
}{}
\iftag{prvOr}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \lor !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\True$ \\
    $\False$ & $\True$ & $\True$ \\
    $\False$ & $\False$ & $\False$ \\
    \hline
  \end{tabular}
}{}%
\iftag{notprvNot,notprvAnd,notprvOr,notprvIf}{}{\\[1em]}
\iftag{prvIf}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \lif !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\False$ \\
    $\False$ & $\True$ & $\True$ \\
    $\False$ & $\False$ & $\True$ \\
    \hline
  \end{tabular}
}{}
\iftag{prvIff}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \liff !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\False$ \\
    $\False$ & $\True$ & $\False$ \\
    $\False$ & $\False$ & $\True$ \\
    \hline
  \end{tabular}
}{}
\end{center}
\end{explain}

\begin{prob}
考虑在 $\Lang{L_0}$ 中加入一个三元联结词 $\diamondsuit$，其求值由下式给出
\begin{gather*}
  \pValue{v}(\diamondsuit( !A , !B , !C ) ) = \begin{cases}
    \pValue{v}( !B ) &
    \text{如果 $\pValue{v}(!A) = \True$;}\\
    \pValue{v}( !C ) &
    \text{如果 $\pValue{v}(!A) = \False $}.
  \end{cases}
\end{gather*}
写出这个联结词的真值表。
\end{prob}

\begin{thm}[局部决定性]
\ollabel{thm:LocalDetermination} 设 $\pAssign{v_1}$ 与
$\pAssign{v_2}$ 是!!{valuation}，它们在 $!A$ 中出现的 !!{propositional
variable}s 上取值一致，即只要 $\Obj p_n$ 出现在某个!!{formula}~$!A$ 中，就有 $\pAssign{v_1}(\Obj p_n) =
\pAssign{v_2}(\Obj p_n)$。则 $\pValue{v_1}$ 与 $\pValue{v_2}$ 在~$!A$ 上也一致，即 $\pValue{v_1}(!A) = \pValue{v_2}(!A)$。
\end{thm}

\begin{proof}
对 $!A$ 作归纳。
\end{proof}

\begin{defn}[满足]
\ollabel{defn:satisfaction} 我们可以归纳地定义
  \emph{!!a{formula}~$!A$ 被!!a{valuation}~$\pAssign{v}$ 满足} 这一概念，记为 $\pSat{v}{!A}$：
  （我们以 $\pSat/{v}{!A}$ 表示「不是 $\pSat{v}{!A}$」。）
\begin{enumerate}
\tagitem{prvFalse}{%
  \indcase{!A}{\lfalse}{$\pSat/{v}{\indfrm}$.}}{}

\tagitem{prvTrue}{%
  \indcase{!A}{\ltrue}{$\pSat{v}{\indfrm}$.}}{}

\item \indcase{!A}{\Obj p_i}{$\pSat{v}{\indfrm}$
  iff $\pAssign{v}(\Obj p_i) = \True$.}

\tagitem{prvNot}{%
  \indcase{!A}{\lnot !B}{$\pSat{v}{\indfrm}$ iff
    $\pSat/{v}{!B}$.}}{}

\tagitem{prvAnd}{%
  \indcase{!A}{(!B \land !C)}{$\pSat{v}{\indfrm}$ iff $\pSat{v}{!B}$
    and $\pSat{v}{!C}$.}}{}

\tagitem{prvOr}{%
  \indcase{!A}{(!B \lor !C)}{$\pSat{v}{\indfrm}$ iff
    $\pSat{v}{!B}$ 或 $\pSat{v}{!C}$ (or both).}}{}

\tagitem{prvIf}{%
  \indcase{!A}{(!B \lif !C)}{$\pSat{v}{\indfrm}$ iff $\pSat/{v}{!B}$
    or $\pSat{v}{!C}$ (or both).}}{}

\tagitem{prvIff}{%
  \indcase{!A}{(!B \liff !C)}{$\pSat{v}{\indfrm}$ iff either both
    $\pSat{v}{!B}$ 且 $\pSat{v}{!C}$, or neither $\pSat{v}{!B}$
    nor $\pSat{v}{!C}$.}}{}
\end{enumerate}
若 $\Gamma$ 是!!{formula}的集合，则 $\pSat{v}{\Gamma}$ 当且仅当对每个~$!A \in \Gamma$ 都有
$\pSat{v}{!A}$。
\end{defn}

\begin{prop}\ollabel{prop:sat-value}
  $\pSat{v}{!A}$ 当且仅当 $\pValue{v}(!A) = \True$。
\end{prop}

\begin{proof}
  对~$!A$ 作归纳。
\end{proof}

\begin{prob}
  证明 \olref[pl][syn][val]{prop:sat-value}
\end{prob}
